import csv
import os
import random
import sys
from tkinter import END, BOTH, StringVar, Text, filedialog, messagebox, ttk
from tkinter import Tk

APP_TITLE = "Lotto Pro Analyse"
NUMBER_MIN = 1
NUMBER_MAX = 49


def generate_demo_history(draw_count=250):
    """Erstellt Demo-Ziehungen ohne externe Daten."""
    random_seed = random.Random(42)
    draws = []
    for _ in range(draw_count):
        draws.append(sorted(random_seed.sample(range(NUMBER_MIN, NUMBER_MAX + 1), 6)))
    return draws


def csv_to_draws(path):
    """Liest CSV-Dateien mit Lotto-Ziehungen ein."""
    draws = []
    with open(path, "r", encoding="utf-8", newline="") as handle:
        reader = csv.reader(handle)
        for row in reader:
            values = []
            for cell in row:
                for part in cell.replace(";", ",").replace(" ", ",").split(","):
                    text = part.strip()
                    if not text:
                        continue
                    try:
                        value = int(text)
                    except ValueError:
                        continue
                    if NUMBER_MIN <= value <= NUMBER_MAX:
                        values.append(value)
            if 6 <= len(values) <= 12:
                unique = sorted(set(values))
                if len(unique) >= 6:
                    draws.append(unique[:6])
    if not draws:
        raise ValueError("Keine gültigen Lotto-Ziehungen in der CSV gefunden.")
    return draws


def compute_number_stats(draws):
    stats = {n: {"count": 0, "last_draw_index": None, "last_gap": None} for n in range(NUMBER_MIN, NUMBER_MAX + 1)}
    for index, draw in enumerate(draws):
        for number in draw:
            previous = stats[number]["last_draw_index"]
            stats[number]["count"] += 1
            if previous is not None:
                stats[number]["last_gap"] = index - previous
            stats[number]["last_draw_index"] = index
    return stats


def build_weighted_pool(stats, total_draws):
    pool = {}
    for number in range(NUMBER_MIN, NUMBER_MAX + 1):
        count = stats[number]["count"]
        gap = stats[number]["last_gap"] if stats[number]["last_gap"] is not None else total_draws
        score = (count * 2.0) + max(0, (total_draws - gap) * 0.7)
        pool[number] = score
    return pool


def valid_set(numbers, min_sum, max_sum, min_even, max_even, min_odd, max_odd):
    if not min_sum <= sum(numbers) <= max_sum:
        return False
    even_count = sum(1 for value in numbers if value % 2 == 0)
    odd_count = 6 - even_count
    if not (min_even <= even_count <= max_even):
        return False
    if not (min_odd <= odd_count <= max_odd):
        return False
    return True


def generate_tips(draws, count, min_sum, max_sum, min_even, max_even, min_odd, max_odd):
    stats = compute_number_stats(draws)
    pool = build_weighted_pool(stats, len(draws))
    number_pool = list(range(NUMBER_MIN, NUMBER_MAX + 1))
    generated = []
    attempts = 0

    while len(generated) < count and attempts < count * 800:
        attempts += 1
        if random.random() < 0.7:
            candidate = []
            while len(candidate) < 6:
                choice = random.choices(number_pool, weights=[pool[n] for n in number_pool], k=1)[0]
                if choice not in candidate:
                    candidate.append(choice)
            candidate = sorted(candidate)
        else:
            candidate = sorted(random.sample(number_pool, 6))

        if valid_set(candidate, min_sum, max_sum, min_even, max_even, min_odd, max_odd):
            generated.append(candidate)

    # Fallback, damit immer gültige Tipps entstehen.
    while len(generated) < count:
        fallback = sorted(random.sample(number_pool, 6))
        if valid_set(fallback, min_sum, max_sum, min_even, max_even, min_odd, max_odd):
            generated.append(fallback)

    return generated[:count]


def format_tip(tip):
    return " ".join(f"{value:02d}" for value in tip)


def export_tips_to_csv(path, tips):
    with open(path, "w", encoding="utf-8", newline="") as handle:
        writer = csv.writer(handle)
        writer.writerow(["Tipp", "Zahlen", "Summe", "Gerade", "Ungerade"])
        for index, tip in enumerate(tips, start=1):
            writer.writerow([
                index,
                format_tip(tip),
                sum(tip),
                sum(1 for value in tip if value % 2 == 0),
                sum(1 for value in tip if value % 2 != 0),
            ])


class LottoApp(Tk):
    def __init__(self):
        super().__init__()
        self.title(APP_TITLE)
        self.geometry("1100x700")
        self.minsize(980, 620)

        self.style = ttk.Style(self)
        self.style.theme_use("clam")
        self.style.configure("Card.TFrame", background="#f4f7fb")
        self.style.configure("Title.TLabel", font=("Segoe UI", 16, "bold"), background="#f4f7fb")
        self.style.configure("Stat.TLabel", font=("Segoe UI", 10, "bold"), background="#f4f7fb")

        self.draws = generate_demo_history(220)
        self.history_path = ""
        self.create_widgets()
        self.refresh_summary()

    def create_widgets(self):
        self.configure(bg="#eef3f8")
        main = ttk.Frame(self, padding=16)
        main.pack(fill="both", expand=True)
        main.columnconfigure(1, weight=1)

        header = ttk.Label(main, text="Lotto Pro Analyse - 6 aus 49", style="Title.TLabel")
        header.grid(row=0, column=0, columnspan=3, sticky="w", pady=(0, 12))

        info = ttk.Label(
            main,
            text="Hinweis: Historische Daten helfen bei Analyse und Organisation. Lotto-Ergebnisse bleiben zufällig und nicht vorhersehbar.",
            foreground="#475569",
            wraplength=820,
            justify="left",
        )
        info.grid(row=1, column=0, columnspan=3, sticky="w", pady=(0, 10))

        actions = ttk.LabelFrame(main, text="Daten")
        actions.grid(row=2, column=0, columnspan=3, sticky="ew", pady=(0, 12))
        actions.columnconfigure(0, weight=1)

        ttk.Button(actions, text="Demo-Daten laden", command=self.load_demo_data).grid(row=0, column=0, sticky="w", padx=10, pady=10)
        ttk.Button(actions, text="CSV importieren", command=self.load_csv).grid(row=0, column=1, sticky="w", padx=10, pady=10)
        self.status_var = StringVar(value="Demo-Daten aktiv")
        ttk.Label(actions, textvariable=self.status_var, foreground="#0f766e").grid(row=0, column=2, sticky="w", padx=10)

        left = ttk.LabelFrame(main, text="Tipp-Generator")
        left.grid(row=3, column=0, sticky="nsew", padx=(0, 12), pady=(0, 10))

        fields = [
            ("Anzahl Tipps", "tip_count", 10),
            ("Min. Summe", "min_sum", 90),
            ("Max. Summe", "max_sum", 150),
            ("Min. Gerade", "min_even", 2),
            ("Max. Gerade", "max_even", 4),
            ("Min. Ungerade", "min_odd", 2),
            ("Max. Ungerade", "max_odd", 4),
        ]
        self.input_vars = {}
        for row_index, (label_text, key, default) in enumerate(fields):
            ttk.Label(left, text=label_text).grid(row=row_index, column=0, sticky="w", padx=10, pady=7)
            var = StringVar(value=str(default))
            self.input_vars[key] = var
            ttk.Entry(left, textvariable=var, width=18).grid(row=row_index, column=1, sticky="w", padx=(0, 10), pady=7)

        ttk.Button(left, text="Tipps erzeugen", command=self.generate).grid(row=len(fields), column=0, sticky="w", padx=10, pady=(12, 8))
        ttk.Button(left, text="CSV exportieren", command=self.export_csv).grid(row=len(fields), column=1, sticky="w", pady=(12, 8))

        right = ttk.LabelFrame(main, text="Statistik")
        right.grid(row=3, column=1, sticky="nsew", pady=(0, 10))
        right.columnconfigure(0, weight=1)

        self.summary_text = Text(right, height=18, width=34, wrap="word", state="disabled", bg="#f8fafc")
        self.summary_text.grid(row=0, column=0, sticky="nsew", padx=10, pady=10)

        table_frame = ttk.LabelFrame(main, text="Erzeugte Tipps")
        table_frame.grid(row=4, column=0, columnspan=2, sticky="nsew", pady=(10, 0))
        table_frame.columnconfigure(0, weight=1)
        table_frame.rowconfigure(0, weight=1)

        self.tree = ttk.Treeview(
            table_frame,
            columns=("tip", "zahlen", "summe", "gerade", "ungerade"),
            show="headings",
            height=12,
        )
        self.tree.heading("tip", text="Tipp Nr.")
        self.tree.heading("zahlen", text="Zahlen")
        self.tree.heading("summe", text="Summe")
        self.tree.heading("gerade", text="Gerade")
        self.tree.heading("ungerade", text="Ungerade")
        self.tree.column("tip", width=90, anchor="center")
        self.tree.column("zahlen", width=260, anchor="center")
        self.tree.column("summe", width=90, anchor="center")
        self.tree.column("gerade", width=90, anchor="center")
        self.tree.column("ungerade", width=100, anchor="center")
        self.tree.grid(row=0, column=0, sticky="nsew", padx=8, pady=8)

    def _read_int(self, key, default):
        try:
            return int(self.input_vars[key].get())
        except ValueError:
            return default

    def load_demo_data(self):
        self.draws = generate_demo_history(220)
        self.status_var.set("Demo-Daten aktiv")
        self.refresh_summary()

    def load_csv(self):
        path = filedialog.askopenfilename(title="CSV-Datei auswählen", filetypes=[("CSV", "*.csv"), ("Alle Dateien", "*.*")])
        if not path:
            return
        try:
            self.draws = csv_to_draws(path)
            self.history_path = path
            self.status_var.set(f"CSV geladen: {os.path.basename(path)}")
            self.refresh_summary()
        except Exception as exc:
            messagebox.showerror("Fehler", f"CSV konnte nicht gelesen werden: {exc}")

    def refresh_summary(self):
        stats = compute_number_stats(self.draws)
        top = sorted(stats.items(), key=lambda item: (item[1]["count"], -item[0]), reverse=True)[:10]
        lines = [
            f"Anzahl Ziehungen: {len(self.draws)}",
            f"Zahlenbereich: {NUMBER_MIN}-{NUMBER_MAX}",
            "",
            "Top häufige Zahlen:",
        ]
        for number, info in top:
            lines.append(f"  {number:02d}: {info['count']}x")
        self._set_summary("\n".join(lines))

    def _set_summary(self, text):
        self.summary_text.configure(state="normal")
        self.summary_text.delete(1.0, END)
        self.summary_text.insert(END, text)
        self.summary_text.configure(state="disabled")

    def generate(self):
        try:
            count = max(1, self._read_int("tip_count", 10))
            min_sum = self._read_int("min_sum", 90)
            max_sum = self._read_int("max_sum", 150)
            min_even = self._read_int("min_even", 2)
            max_even = self._read_int("max_even", 4)
            min_odd = self._read_int("min_odd", 2)
            max_odd = self._read_int("max_odd", 4)
        except Exception:
            messagebox.showerror("Fehler", "Bitte nur ganze Zahlen eingeben.")
            return

        if not self.draws:
            messagebox.showerror("Fehler", "Bitte erst Ziehungsdaten laden.")
            return

        tips = generate_tips(self.draws, count, min_sum, max_sum, min_even, max_even, min_odd, max_odd)
        self.tree.delete(*self.tree.get_children())
        for index, tip in enumerate(tips, start=1):
            self.tree.insert(
                "",
                "end",
                values=(
                    index,
                    format_tip(tip),
                    sum(tip),
                    sum(1 for value in tip if value % 2 == 0),
                    sum(1 for value in tip if value % 2 != 0),
                ),
            )

    def export_csv(self):
        rows = [self.tree.item(item, "values") for item in self.tree.get_children()]
        if not rows:
            messagebox.showwarning("Warnung", "Es gibt noch keine Tipps zum Exportieren.")
            return

        file_path = filedialog.asksaveasfilename(
            title="Tipps als CSV speichern",
            defaultextension=".csv",
            filetypes=[("CSV Datei", "*.csv"), ("Alle Dateien", "*.*")],
        )
        if not file_path:
            return

        try:
            tips = []
            for row in rows:
                nums = [int(value.strip()) for value in row[1].split()]
                tips.append(nums)
            export_tips_to_csv(file_path, tips)
            messagebox.showinfo("Erfolg", f"Tipps wurden gespeichert:\n{file_path}")
        except Exception as exc:
            messagebox.showerror("Fehler", f"Export fehlgeschlagen: {exc}")


if __name__ == "__main__":
    app = LottoApp()
    app.mainloop()
    sys.exit(0)
